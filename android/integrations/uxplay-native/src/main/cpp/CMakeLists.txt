cmake_minimum_required(VERSION 3.22.1)
project(tvcompanion_uxplay LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../.native" ABSOLUTE)
set(UX "${DEPS}/uxplay")
set(PLIST "${DEPS}/libplist")
if(NOT EXISTS "${DEPS}/openssl-armv7/lib/libcrypto.a")
    message(FATAL_ERROR "Run scripts/prepare-airplay-native.py first")
endif()
file(GLOB PLIST_SRC CONFIGURE_DEPENDS "${PLIST}/src/*.c")
add_library(plist STATIC ${PLIST_SRC} "${PLIST}/libcnary/node.c" "${PLIST}/libcnary/node_list.c")
target_include_directories(plist PUBLIC "${PLIST}/include" PRIVATE "${PLIST}/libcnary/include" "${PLIST}" "${PLIST}/src")
target_compile_definitions(plist PRIVATE PACKAGE_VERSION="2.7.0" HAVE_MEMMEM=1 HAVE_STRNDUP=1 HAVE_LOCALTIME_R=1 HAVE_GMTIME_R=1 HAVE_TIMEGM=1 HAVE_STRPTIME=1 HAVE_TM_TM_GMTOFF=1 HAVE_TM_TM_ZONE=1)
file(GLOB PLAYFAIR_SRC CONFIGURE_DEPENDS "${UX}/lib/playfair/*.c")
add_library(playfair STATIC ${PLAYFAIR_SRC})
target_include_directories(playfair PUBLIC "${UX}/lib/playfair")
file(GLOB LLHTTP_SRC CONFIGURE_DEPENDS "${UX}/lib/llhttp/*.c")
add_library(llhttp STATIC ${LLHTTP_SRC})
target_include_directories(llhttp PUBLIC "${UX}/lib/llhttp")
file(GLOB MDNS_SRC CONFIGURE_DEPENDS "${UX}/lib/mdnsd/*.c")
add_library(dnssd STATIC ${MDNS_SRC})
target_include_directories(dnssd PUBLIC "${UX}/lib/mdnsd" PRIVATE "${UX}/lib")
file(GLOB AIRPLAY_SRC CONFIGURE_DEPENDS "${UX}/lib/*.c")
add_library(receiver STATIC ${AIRPLAY_SRC})
target_include_directories(receiver PUBLIC "${UX}/lib" PRIVATE "${DEPS}/openssl-armv7/include")
target_compile_definitions(receiver PRIVATE OPENSSL_API_COMPAT=0x10101000L PLIST_210=1 PLIST_230=1)
target_link_libraries(receiver PRIVATE plist playfair llhttp dnssd "${DEPS}/openssl-armv7/lib/libcrypto.a")
set(ALAC "${DEPS}/alac/codec")
add_library(alac STATIC "${ALAC}/ALACDecoder.cpp" "${ALAC}/ALACBitUtilities.c" "${ALAC}/EndianPortable.c" "${ALAC}/ag_dec.c" "${ALAC}/dp_dec.c" "${ALAC}/matrix_dec.c")
target_include_directories(alac PUBLIC "${ALAC}")
add_library(tvcompanion_uxplay SHARED receiver_jni.cpp audio_output.cpp)
target_link_libraries(tvcompanion_uxplay PRIVATE receiver alac android mediandk aaudio log)
